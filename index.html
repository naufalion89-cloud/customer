Sejarah Singkat Perusahaan

PT. JALUR DATA INDONESIA didirikan berawal dari tekad kuat untuk menghadirkan akses internet fiber optic berkualitas tinggi yang merata. Kami melihat tingginya kebutuhan masyarakat akan internet cepat, stabil, dan terjangkau di kawasan Sukoharjo, Banyumas, hingga BANJAREJO, BUMI ARUM, PRINGSEWU dan Pantura. Melalui investasi infrastruktur jalur kabel optic mandiri dan perangkat berstandar industri, PT. JDI kini menjadi mitra terpercaya bagi ribuan rumah tangga, sekolah, instansi balai desa, dan UMKM.

Kode Google Apps Script (Code.gs)

Gunakan kode di bawah ini pada Editor Skrip di Spreadsheet Anda (1bR4QVxYM1sTtdaNgCS3uNA9vQHuw7iCL2cXu_DDN3TM) lalu deploy sebagai Web App (Akses: Anyone):

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var ss = SpreadsheetApp.openById("1bR4QVxYM1sTtdaNgCS3uNA9vQHuw7iCL2cXu_DDN3TM");
    var sheet = ss.getSheetByName("Pelanggan_Baru") || ss.insertSheet("Pelanggan_Baru");
    
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["Timestamp", "No. Reservasi", "Nama", "No. WhatsApp", "Desa", "Paket", "Alamat", "Status", "Foto Lokasi"]);
    }
    
    var data = JSON.parse(e.postData.contents);
    sheet.appendRow([
      new Date(),
      data.resNumber,
      data.name,
      data.phone,
      data.desa,
      data.package,
      data.address,
      data.status || "Menunggu Verifikasi",
      data.photoData || "-"
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({result: "success", resNumber: data.resNumber}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({result: "error", error: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function doGet() {
  return ContentService.createTextOutput(JSON.stringify({status: "online", isp: "PT. JALUR DATA INDONESIA"}))
    .setMimeType(ContentService.MimeType.JSON);
}
